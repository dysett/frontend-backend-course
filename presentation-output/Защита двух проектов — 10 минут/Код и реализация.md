```javascript
async function api(path, { method = 'GET', body, inspect = true } = {}) {
  let response;
  let json;
  try {
    response = await fetch(path, {
      method, signal: AbortSignal.timeout(12000),
      headers: { Accept: 'application/json', ...(body === undefined ? {} : { 'Content-Type': 'application/json' }) },
      ...(body === undefined ? {} : { body: JSON.stringify(body) }),
    });
    json = await response.json();
  } catch {
    if (inspect) recordRequest(method, path, body, 'Немає відповіді', { error: 'Не вдалося отримати JSON-відповідь сервера.' });
    throw new Error('Немає зв’язку із сервісом. Перевірте, що сервер запущений, і повторіть дію.');
  }
  if (inspect) recordRequest(method, path, body, response.status, json);
  if (!response.ok) throw new Error(json.error?.message || 'Не вдалося виконати дію.');
  return json;
}
```

```javascript
function object(value, allowed) {
  if (!value || typeof value !== 'object' || Array.isArray(value)) throw invalid('Тіло запиту має бути JSON-об’єктом.');
  const unknown = Object.keys(value).filter(key => !allowed.includes(key));
  if (unknown.length) throw invalid(`Невідомі поля: ${unknown.join(', ')}.`);
}

function text(value, name, max) {
  if (typeof value !== 'string' || !value.trim() || value.trim().length > max) {
    throw invalid(`${name}: потрібен непорожній рядок довжиною до ${max} символів.`);
  }
  return value.trim();
}

function integer(value, name, min, max) {
  if (!Number.isInteger(value) || value < min || value > max) throw invalid(`${name}: потрібне ціле число від ${min} до ${max}.`);
  return value;
}

function isbn13(value) {
  const isbn = text(value, 'isbn', 32).replace(/[ -]/g, '');
  if (!/^97[89]\d{10}$/.test(isbn)) throw invalid('isbn: потрібен ISBN-13 із префіксом 978 або 979.');
  const sum = [...isbn].reduce((total, digit, i) => total + Number(digit) * (i % 2 ? 3 : 1), 0);
  if (sum % 10 !== 0) throw invalid('isbn: неправильна контрольна цифра ISBN-13.');
  return isbn;
}

export function bookInput(body, { partial = false, now = new Date() } = {}) {
  const fields = ['title', 'author', 'isbn', 'publicationYear', 'genre', 'totalCopies'];
  object(body, fields);
  if (!partial && fields.some(field => !Object.hasOwn(body, field))) throw invalid(`Обов’язкові поля: ${fields.join(', ')}.`);
  if (partial && !Object.keys(body).length) throw invalid('Передайте хоча б одне поле для оновлення.');
  const result = {};
  for (const field of fields) {
    if (!Object.hasOwn(body, field)) continue;
    if (field === 'isbn') result[field] = isbn13(body[field]);
    else if (field === 'publicationYear') result[field] = integer(body[field], field, 1, now.getUTCFullYear());
    else if (field === 'totalCopies') result[field] = integer(body[field], field, 1, 100);
    else result[field] = text(body[field], field, field === 'genre' ? 80 : 200);
  }
  return result;
```

```javascript
export async function api(path, { method = 'GET', body } = {}) {
  let response;
  try { response = await fetch(`/api${path}`, {
    method, credentials: 'same-origin', cache: 'no-store', signal: AbortSignal.timeout(8000),
    headers: { Accept: 'application/json', ...(body !== undefined ? { 'Content-Type': 'application/json' } : {}),
      ...(!['GET', 'HEAD'].includes(method) ? { 'X-CSRF-Token': csrfToken ?? '' } : {}) },
    body: body === undefined ? undefined : JSON.stringify(body),
  }); } catch {
    throw new Error('Немає зв’язку із сервером. Перевірте підключення та повторіть дію.');
  }
  const json = await response.json();
  if (!response.ok) {
    const error = new Error(json.error?.message ?? 'Не вдалося виконати запит.');
    error.code = json.error?.code;
    error.status = response.status;
    if (response.status === 401 && path !== '/auth/login') window.dispatchEvent(new Event('session-expired'));
    throw error;
  }
  if (json.data?.csrfToken) csrfToken = json.data.csrfToken;
  return json.data;
}
```

```javascript
  if (typeof password !== 'string' || password.length < 12 || password.length > 128) throw invalid('Пароль має містити від 12 до 128 символів.');
  const salt = randomBytes(16).toString('hex');
```

```javascript
export function requireAdmin(req, res, next) {
  if (req.session?.user?.role !== 'admin') return next(new AppError(401, 'AUTH_REQUIRED', 'Увійдіть як адміністратор.'));
  next();
}
export function csrfProtection(req, res, next) {
  if (safeMethod(req.method)) return next();
  const token = req.get('X-CSRF-Token');
  const expected = req.session?.csrfToken;
  if (typeof token !== 'string' || !expected || Buffer.byteLength(token) !== Buffer.byteLength(expected) || !timingSafeEqual(Buffer.from(token), Buffer.from(expected))) {
    return next(new AppError(403, 'CSRF_INVALID', 'Сеанс перевірки застарів. Оновіть сторінку й повторіть дію.'));
  }
  next();
}
export function sameOrigin(publicOrigin) {
  return (req, res, next) => {
    const host = publicOrigin ? new URL(publicOrigin).hostname : null;
    if (!(host ? req.hostname === host : ['127.0.0.1', 'localhost', '[::1]'].includes(req.hostname))) return next(new AppError(403, 'HOST_FORBIDDEN', 'Недозволений хост. Налаштуйте PUBLIC_ORIGIN.'));
    if (!safeMethod(req.method)) {
      const expected = publicOrigin || `${req.protocol}://${req.get('host')}`;
      if ((req.get('Origin') && req.get('Origin') !== expected) || req.get('Sec-Fetch-Site') === 'cross-site') return next(new AppError(403, 'ORIGIN_FORBIDDEN', 'Запити з іншого сайту заборонені.'));
    }
    next();
  };
}
```
